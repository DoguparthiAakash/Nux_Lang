145: func lex_is_digit(c) {
146:     if (c >= '0') {
147:         if (c <= '9') {
148:             return 1;
149:         }
150:     }
151:     return 0;
152: }

154: func lex_is_alpha(c) {
155:     if (c >= 'a') {
156:         if (c <= 'z') { return 1; }
157:     }
158:     if (c >= 'A') {
159:         if (c <= 'Z') { return 1; }
160:     }
161:     if (c == '_') { return 1; }
162:     return 0;
163: }

165: func lex_is_alnum(c) {
166:     if (lex_is_digit(c) == 1) { return 1; }
167:     if (lex_is_alpha(c) == 1) { return 1; }
168:     return 0;
169: }

171: func lex_is_hex(c) {

175:     }
176:     if (c >= 'A') {

687:             return 0;

690:         var c = this.cur();
691:

693:         if (c == '-') { this.advance_pos(); this.set_tk(TK_MINUS); return 0; }
694:         if (c == '%') { this.advance_pos(); this.set_tk(TK_PERCENT); return 0; }
695:         if (c == '(') { this.advance_pos(); this.set_tk(TK_LPAREN); return 0; }

698:         if (c == ']') { this.advance_pos(); this.set_tk(TK_RBRACKET); return 0; }

700:         if (c == '}') { this.advance_pos(); this.set_tk(TK_RBRACE); return 0; }
701:         if (c == ';') { this.advance_pos(); this.set_tk(TK_SEMICOLON); return 0; }
702:         if (c == '.') { this.advance_pos(); this.set_tk(TK_DOT); return 0; }

707:             this.advance_pos();

709:                 this.advance_pos();

848:             return 0;
849:         }
850: 
851:         if (lex_is_alpha(c) == 1) {
852:             this.lex_identifier();
853:             return 0;
854:         }
855: 
856:         this.advance_pos();
857:         this.set_tk(TK_IDENTIFIER);
858:         return 0;
859:     }
